\pagestyle{empty}

% Draw a panel frame: \panel{x1}{y1}{x2}{y2} (coordinates in cm)
\newcommand{\panel}[4]{%
  \draw[line width=2pt, color=comicblue] (#1,#2) rectangle (#3,#4);%
  \fill[panelgrey] (#1,#2) rectangle (#3,#4);%
}

% Draw a speech bubble: \speech{x}{y}{width}{text}
\newcommand{\speech}[3]{%
  \node[draw=comicblue, line width=1.5pt, fill=white, rounded corners=8pt,
        align=center, text width=#2cm] at (#1) {\textbf{#3}};%
}

% Draw a caption box: \caption{x}{y}{width}{text}
\newcommand{\captionbox}[3]{%
  \node[draw=comicred, line width=1.5pt, fill=comicyellow!30,
        align=left, text width=#2cm] at (#1) {\small #3};%
}
